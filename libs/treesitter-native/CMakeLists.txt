# Static library with the tree-sitter runtime, every vendored grammar and the grammar table.
# Inputs come from `scripts/grammars.py fetch` (build/vendor); driven by Gradle task :treesitter-native:buildGrammars.
cmake_minimum_required(VERSION 3.20)
project(rtok_ts C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(VENDOR ${CMAKE_CURRENT_SOURCE_DIR}/build/vendor)
set(RTOK_TS_GRAMMARS "" CACHE STRING "Semicolon-separated grammar names to build (empty = all)")
add_library(rtok_ts STATIC ${VENDOR}/tree-sitter/lib/src/lib.c ${VENDOR}/rtok_ts_table.c)
target_include_directories(rtok_ts PRIVATE ${VENDOR}/tree-sitter/lib/include ${VENDOR}/tree-sitter/lib/src src/nativeInterop/c)
file(STRINGS ${VENDOR}/grammars.txt rows)
foreach(row IN LISTS rows)
  string(REPLACE "\t" ";" cols "${row}")
  list(GET cols 0 name)
  list(GET cols 1 dir)
  set(srcs ${dir}/parser.c)
  if(EXISTS ${dir}/scanner.c)
    list(APPEND srcs ${dir}/scanner.c)
  endif()
  add_library(g_${name} OBJECT ${srcs})
  target_include_directories(g_${name} PRIVATE ${dir})
  target_compile_options(g_${name} PRIVATE -w)
  target_sources(rtok_ts PRIVATE $<TARGET_OBJECTS:g_${name}>)
endforeach()
target_compile_options(rtok_ts PRIVATE -w)
# Dev-only (`mise run coverage`): source-based coverage of the tree-sitter runtime.
option(RTOK_TS_COVERAGE "Instrument the runtime for llvm-cov" OFF)
if(RTOK_TS_COVERAGE)
  set_source_files_properties(${VENDOR}/tree-sitter/lib/src/lib.c PROPERTIES COMPILE_OPTIONS "-fprofile-instr-generate;-fcoverage-mapping")
endif()
